// Turn the photographed sheet of paper into its light and shadow alone: grey,
// with its uneven lighting taken out (each pixel divided by the broad
// brightness around it), so flat paper sits at one level (173 of 255) with
// folds darker and catches of light brighter. The photograph is a gentle
// one, so its contrast about that level is raised by `gain` (after a slight
// blur, so camera noise is not raised with it; the paper's own grain stays).
//
//   node prepare-paper.js <photograph> <folder to write to>
//
// Needs `sharp`. Not part of the build.
const sharp = require('sharp');
const fs = require('fs');
const LEVEL = 173;
const WIDTH = 2880; // the whole photograph, 1.5 times a 1920 cover
const GAIN = 3.6;
const SOFT = 0.8;

(async () => {
  const [,, src, outDir] = process.argv;
  const { data, info } = await sharp(src, { failOn: 'none' }).rotate().greyscale().resize({ width: WIDTH }).blur(SOFT).raw().toBuffer({ resolveWithObject: true });
  const w = info.width, h = info.height, ch = info.channels; // (sharp may hand back three channels: read the count)
  // the broad brightness: the picture made small, blurred and brought back to size
  const tiny = await sharp(data, { raw: { width: w, height: h, channels: ch } }).resize({ width: Math.round(w / 12) }).blur(10).png().toBuffer();
  const lowOut = await sharp(tiny).resize({ width: w, height: h, fit: 'fill', kernel: 'cubic' }).raw().toBuffer({ resolveWithObject: true });
  const low = lowOut.data, lc = lowOut.info.channels;
  if (lowOut.info.width !== w || lowOut.info.height !== h) throw new Error('size mismatch');
  const o = Buffer.alloc(w * h);
  const hist = new Uint32Array(256);
  for (let i = 0; i < w * h; i += 1) { const v = Math.max(0, Math.min(255, Math.round(LEVEL + ((data[i * ch] / Math.max(1, low[i * lc])) * LEVEL - LEVEL) * GAIN))); o[i] = v; hist[v] += 1; }
  const pct = (p) => { let a = 0; for (let v = 0; v < 256; v += 1) { a += hist[v]; if (a >= w * h * p) return v; } return 255; };
  const sheet = sharp(o, { raw: { width: w, height: h, channels: 1 } });
  await sheet.clone().jpeg({ quality: 84, mozjpeg: true }).toFile(`${outDir}/paper-folds.jpg`);
  await sheet.clone().resize({ width: 480 }).jpeg({ quality: 86, mozjpeg: true }).toFile(`${outDir}/paper-folds-small.jpg`);
  console.log(`${w}x${h}`, 'p1', pct(0.01), 'median', pct(0.5), 'p99', pct(0.99), fs.statSync(`${outDir}/paper-folds.jpg`).size, fs.statSync(`${outDir}/paper-folds-small.jpg`).size);
})();
